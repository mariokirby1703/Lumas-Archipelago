cmake_minimum_required(VERSION 3.20)
project(LCUDeathBridge LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Requires MSVC x64 (SEH and Windows ABI)")
endif()
add_library(LCUDeathBridge SHARED death_bridge.cpp)
target_compile_features(LCUDeathBridge PRIVATE cxx_std_17)
target_compile_definitions(LCUDeathBridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(LCUDeathBridge PRIVATE /W4 /WX /guard:cf)
target_link_libraries(LCUDeathBridge PRIVATE user32)
